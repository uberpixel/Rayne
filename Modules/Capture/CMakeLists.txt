cmake_minimum_required(VERSION 3.28)
project(RayneCapture)

include(../../CMake/LibraryTarget.cmake)

option(RN_CAPTURE_ENABLE_RTMP "Enable the optional RTMP encoded media sink" OFF)

set(SOURCES RNCaptureInit.cpp RNCaptureSession.cpp)
set(HEADERS RNCapture.h RNCaptureSession.h)
set(PRIVATE_HEADERS RNCaptureEncodedSink.h)
set(CAPTURE_LIBRARIES Rayne)
set(CAPTURE_PRIVATE_LIBRARIES)

if(APPLE)
    enable_language(OBJCXX)
    list(APPEND SOURCES RNCaptureAppleSession.mm)
    list(APPEND CAPTURE_LIBRARIES RayneMetal)
    list(APPEND CAPTURE_PRIVATE_LIBRARIES "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo" "-framework Foundation" "-framework Metal" "-framework QuartzCore")
    if(IOS OR VISIONOS)
        list(APPEND CAPTURE_PRIVATE_LIBRARIES "-framework UIKit")
    endif()
    set_source_files_properties(RNCaptureAppleSession.mm PROPERTIES COMPILE_FLAGS "-fno-objc-arc")
elseif(ANDROID)
    list(APPEND SOURCES RNCaptureAndroidSession.cpp RNCaptureAndroidFileWriter.cpp)
    list(APPEND PRIVATE_HEADERS RNCaptureAndroidFileWriter.h)
    list(APPEND CAPTURE_PRIVATE_LIBRARIES mediandk android)
elseif(WIN32)
    list(APPEND SOURCES RNCaptureWindowsSession.cpp RNCaptureWindowsSwapChain.cpp RNCaptureWindowsInternals.cpp RNCaptureWindowsFileWriter.cpp RNCaptureWindowsEncoder.cpp)
    list(APPEND PRIVATE_HEADERS RNCaptureWindowsSwapChain.h RNCaptureWindowsInternals.h RNCaptureWindowsFileWriter.h RNCaptureWindowsEncoder.h)
    list(APPEND CAPTURE_LIBRARIES RayneVulkan)
    list(APPEND CAPTURE_PRIVATE_LIBRARIES d3d11 dxgi mf mfplat mfreadwrite mfuuid)
endif()

if(RN_CAPTURE_ENABLE_RTMP)
    list(APPEND SOURCES RNCaptureRTMPWriter.cpp)
    list(APPEND PRIVATE_HEADERS RNCaptureRTMPWriter.h)
    if(ANDROID)
        if(TARGET ssl AND TARGET crypto)
            list(APPEND CAPTURE_PRIVATE_LIBRARIES ssl crypto)
        else()
            find_package(OpenSSL REQUIRED)
            list(APPEND CAPTURE_PRIVATE_LIBRARIES OpenSSL::SSL OpenSSL::Crypto)
        endif()
    elseif(WIN32)
        list(APPEND CAPTURE_PRIVATE_LIBRARIES ws2_32 secur32)
    endif()
endif()

rayne_add_library(RayneCapture "${SOURCES}" "${HEADERS}" "0.1.0" "1" PUBLIC_LIBRARIES "${CAPTURE_LIBRARIES}" PRIVATE_HEADERS "${PRIVATE_HEADERS}")
rayne_set_module_output_directory(RayneCapture)
if(APPLE)
    set_target_properties(RayneCapture PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
endif()
target_compile_definitions(RayneCapture PRIVATE RN_BUILD_CAPTURE RN_CAPTURE_ENABLE_RTMP=$<BOOL:${RN_CAPTURE_ENABLE_RTMP}>)
if(CAPTURE_PRIVATE_LIBRARIES)
    target_link_libraries(RayneCapture PRIVATE ${CAPTURE_PRIVATE_LIBRARIES})
endif()
